cmake_minimum_required(VERSION 3.16)
project(gear_resolver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_GUI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SC_NO_NETWORKING ON CACHE BOOL "" FORCE)
set(SC_NO_THREADING ON CACHE BOOL "" FORCE)
add_compile_definitions(SC_USE_PTR=0)
if(NOT SIMC_SOURCE_DIR)
  message(FATAL_ERROR "Set SIMC_SOURCE_DIR to the pinned SimulationCraft source directory")
endif()
add_subdirectory("${SIMC_SOURCE_DIR}" simc EXCLUDE_FROM_ALL)
add_executable(gear-resolver gear_resolver.cpp)
target_link_libraries(gear-resolver PRIVATE engine)
